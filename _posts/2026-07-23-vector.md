using Matrix = vector<vector<double>>;

{
	Matrix u(3, vector<double>(3));
	
	u[0][0] = 1;
	u[0][1] = 2;
	u[0][2] = 3;

	u[1][0] = 4;
	u[1][1] = 5;
	u[1][2] = 6;

	u[2][0] = 7;
	u[2][1] = 8;
	u[2][2] = 9;

	for (int i = 0; i < u.size(); i++)
	{
		for (int j = 0; j < u[i].size(); j++)
		{
			cout << u[i][j] << " ";
		}
		cout << endl;
	}

	for (const auto& row : u)
	{
		for (double value : row)
		{
			cout << value << " ";
		}
		cout << endl;
	}

}
